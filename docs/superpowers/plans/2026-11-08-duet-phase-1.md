# Karteek Duet Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers-extended-cc:subagent-driven-development (recommended) or superpowers-extended-cc:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the governed knowledge graph under the Karteek vault: schema, gates, journal, indexer, Markdown views, a read-only MCP server, and rebuild, so the current vault indexes with zero gate failures and Claudian can answer from Claim nodes.

**Architecture:** One Python package `duet/` in this code repo. `graph.py` owns a LadybugDB file in the vault's `private/graph/` and exposes a single gated write path `apply(changeset, actor)` that journals every result. `ecl.py` parses vault pages into changesets, `project.py` renders views to `private/views/`, `mcp_server.py` serves read tools over stdio. The vault is found through `KARTEEK_VAULT`.

**Tech Stack:** Python 3.12 virtual environment at `.venv` (LadybugDB has no 3.14 wheel). `ladybug==0.22.3` (import name `ladybug`; `Database`, `Connection`, `execute(stmt, parameters=dict)`), `mcp==2.3.0` (`from mcp.server.mcpserver import MCPServer`). Tests: standard-library `unittest`. All test commands below use `.venv/bin/python`.

**Spec:** the vault's `docs/specs/2026-11-08-karteek-duet-design.md`. Phase 1 rows of its phasing table. Phase 2 (runner, daily note, write tools) and phase 3 are separate plans.

**Repository rules:** code only in this repo; never copy vault pages here. Commits happen because Karteek approved executing this plan; each task ends in one commit. The existing `scripts/` and their eleven tests stay untouched and must keep passing.

**Deviations from the spec, decided here:** Claim ids derive from page path plus heading slug (`claim:identity/profile:professional-role`), not from the normalized statement, so a reworded claim updates in place with history instead of creating an orphan. Only sections with a `**Claim:**` field are Claims; the dated entries in `memory/current.md` are Documents only in phase 1. `search` uses case-insensitive substring match over labels; the full-text index comes with phase 3.

```python
"""Locate the knowledge vault and derive every path the code writes to."""
import os
from dataclasses import dataclass
from pathlib import Path

CODE_ROOT = Path(__file__).resolve().parents[1]


def vault_path() -> Path:
    """Return the vault directory from KARTEEK_VAULT or the code repo's .env file."""
    value = os.environ.get('KARTEEK_VAULT')
    if not value:
        env = CODE_ROOT / '.env'
        if env.is_file():
            for line in env.read_text().splitlines():
                line = line.strip()
                if line.startswith('KARTEEK_VAULT='):
                    value = line.split('=', 1)[1].strip().strip('"').strip("'")
                    break
    if not value:
        raise SystemExit('Set KARTEEK_VAULT (env or .env) to the vault directory. See .env.example.')
    path = Path(value).expanduser()
    if not (path / 'AGENTS.md').is_file():
        raise SystemExit(f'{path} does not look like the Karteek vault (no AGENTS.md).')
    return path


@dataclass(frozen=True)
class Paths:
    vault: Path

    @property
    def graph_dir(self) -> Path:
        return self.vault / 'private' / 'graph'

    @property
    def db(self) -> Path:
        return self.graph_dir / 'duet.lbug'

    @property
    def journal(self) -> Path:
        return self.graph_dir / 'journal.jsonl'

    @property
    def index_state(self) -> Path:
        return self.graph_dir / 'index-state.json'

    @property
    def index_report(self) -> Path:
        return self.graph_dir / 'index-report.json'

    @property
    def lock(self) -> Path:
        return self.graph_dir / 'duet.lock'

    @property
    def views(self) -> Path:
        return self.vault / 'private' / 'views'


def paths(vault=None) -> Paths:
    return Paths(Path(vault).expanduser() if vault else vault_path())
```

```python
"""Node and relation tables of the Duet graph: the Model step."""
COMMON = ('id STRING PRIMARY KEY, label STRING, scope STRING, owner STRING, created STRING, '
          'updated STRING, source_locator STRING, defined_in STRING, history STRING[]')

NODE_TABLES = {
    'Person': 'is_karteek BOOLEAN, role STRING',
    'Claim': 'status STRING, scope_text STRING, recorded STRING, review_by STRING, exceptions STRING',
    'Source': 'kind STRING, date STRING, origin STRING, provider STRING',
    'Feedback': 'date STRING, scope_text STRING, applied BOOLEAN',
    'Account': 'stage STRING',
    'Meeting': 'start STRING, provider_ids STRING[], summary_hash STRING',
    'Decision': 'date STRING, rationale STRING, reopen_when STRING',
    'Commitment': 'due STRING, status STRING, owner_known BOOLEAN',
    'Concept': 'author STRING',
    'Document': 'path STRING, hash STRING, frontmatter_type STRING',
    'Playbook': 'area STRING, execution STRING',
    'DuetItem': 'key STRING, revision STRING, status STRING, snooze_until STRING, why_now STRING, draft STRING',
}

REL_TABLES = {
    'SUPPORTED_BY': [('Claim', 'Source'), ('Decision', 'Source'), ('Commitment', 'Source'), ('Meeting', 'Source')],
    'SUPERSEDES': [('Claim', 'Claim')],
    'CORRECTS': [('Feedback', 'Claim')],
    'ABOUT': [('Claim', 'Person'), ('Claim', 'Account')],
    'ATTENDED': [('Person', 'Meeting')],
    'DECIDED_IN': [('Decision', 'Meeting')],
    'COMMITTED_BY': [('Commitment', 'Person')],
    'CONCERNS': [('Meeting', 'Account'), ('Decision', 'Account'), ('Commitment', 'Account')],
    'MENTIONS': [('Document', table) for table in NODE_TABLES],
    'LINKS_TO': [('Document', 'Document')],
    'ATTRIBUTED_TO': [('Concept', 'Person')],
    'DERIVED_FROM': [('Concept', 'Document')],
    'USES': [('DuetItem', 'Playbook')],
    'RESTS_ON': [('DuetItem', 'Meeting'), ('DuetItem', 'Commitment'), ('DuetItem', 'Decision'), ('DuetItem', 'Claim')],
}
REL_PROPS = 'locator STRING, created STRING'

STATUSES = ('confirmed', 'hypothesis', 'unknown', 'superseded')
EVIDENCE_TABLES = ('Claim', 'Decision', 'Commitment', 'Meeting')
ACTORS = ('indexer', 'mcp:claude')
SCOPES = ('tracked', 'private')


def ddl():
    """Idempotent CREATE statements for every table."""
    statements = [f'CREATE NODE TABLE IF NOT EXISTS {name}({COMMON}, {props})' for name, props in NODE_TABLES.items()]
    for name, pairs in REL_TABLES.items():
        froms = ', '.join(f'FROM {a} TO {b}' for a, b in pairs)
        statements.append(f'CREATE REL TABLE IF NOT EXISTS {name}({froms}, {REL_PROPS})')
    return statements
```

```python
"""The one owner of the LadybugDB file: open, schema, reads, and the gated write path."""
import fcntl
import json
from datetime import datetime, timezone

import ladybug as lb

from duet.schema import NODE_TABLES, ddl


def now_iso():
    return datetime.now(timezone.utc).isoformat(timespec='seconds')


class Graph:
    def __init__(self, p, db, conn, lock_file):
        self.paths = p
        self._db = db
        self._conn = conn
        self._lock_file = lock_file

    @classmethod
    def open(cls, p, read_only=False):
        p.graph_dir.mkdir(parents=True, exist_ok=True)
        lock_file = None
        if not read_only:
            lock_file = p.lock.open('a')
            try:
                fcntl.flock(lock_file, fcntl.LOCK_EX | fcntl.LOCK_NB)
            except BlockingIOError:
                lock_file.close()
                raise RuntimeError('another duet process holds the graph')
        db = lb.Database(str(p.db), read_only=read_only)
        return cls(p, db, lb.Connection(db), lock_file)

    def close(self):
        self._conn.close()
        self._db.close()
        if self._lock_file:
            fcntl.flock(self._lock_file, fcntl.LOCK_UN)
            self._lock_file.close()
            self._lock_file = None

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

    # reads
    def execute(self, statement, params=None):
        return self._conn.execute(statement, parameters=params) if params else self._conn.execute(statement)

    def rows(self, statement, params=None):
        result = self.execute(statement, params)
        out = []
        while result.has_next():
            out.append(result.get_next())
        return out

    def ensure_schema(self):
        for statement in ddl():
            self.execute(statement)

    def tables(self):
        return [row[1] for row in self.rows('CALL SHOW_TABLES() RETURN *')]

    def node(self, node_id):
        found = self.rows('MATCH (n {id: $id}) RETURN n', {'id': node_id})
        if not found:
            return None
        raw = found[0][0]
        return raw['_LABEL'], {k: v for k, v in raw.items() if not k.startswith('_')}

    def neighbors(self, node_id, rel=None, direction='both'):
        out = []
        if direction in ('out', 'both'):
            out += self.rows('MATCH (n {id: $id})-[r]->(m) RETURN label(r), label(m), m.id, r.locator ORDER BY m.id', {'id': node_id})
        if direction in ('in', 'both'):
            out += self.rows('MATCH (n {id: $id})<-[r]-(m) RETURN label(r), label(m), m.id, r.locator ORDER BY m.id', {'id': node_id})
        tuples = [tuple(row) for row in out]
        return [t for t in tuples if rel is None or t[0] == rel]

    def counts(self):
        return {table: self.rows(f'MATCH (n:{table}) RETURN count(n)')[0][0] for table in NODE_TABLES}

    # journal
    def journal(self, result, actor, changeset, when, gate=None, message=''):
        entry = {'ts': when, 'actor': actor, 'result': result, 'gate': gate, 'message': message,
                 'changeset': changeset.to_json() if actor != 'indexer' else changeset.summary()}
        with self.paths.journal.open('a') as stream:
            stream.write(json.dumps(entry, ensure_ascii=False) + '\n')
```

```python
"""The unit of writing: nodes to upsert, edges to merge, node ids to retire."""
from dataclasses import asdict, dataclass, field


class GateError(Exception):
    def __init__(self, gate, message):
        super().__init__(f'{gate}: {message}')
        self.gate = gate


@dataclass
class Node:
    table: str
    id: str
    props: dict = field(default_factory=dict)


@dataclass
class Edge:
    rel: str
    src: str
    dst: str
    locator: str = ''


@dataclass
class Changeset:
    nodes: list = field(default_factory=list)
    edges: list = field(default_factory=list)
    retire: list = field(default_factory=list)

    def to_json(self):
        return {'nodes': [asdict(n) for n in self.nodes], 'edges': [asdict(e) for e in self.edges], 'retire': list(self.retire)}

    @classmethod
    def from_json(cls, data):
        return cls(nodes=[Node(**n) for n in data.get('nodes', [])],
                   edges=[Edge(**e) for e in data.get('edges', [])],
                   retire=list(data.get('retire', [])))

    def summary(self):
        return {'nodes': [n.id for n in self.nodes], 'edges': len(self.edges), 'retire': list(self.retire)}
```

```python
"""The Constrain and Validate steps: a changeset is applied whole or rejected whole."""
from collections import defaultdict

from duet.changeset import GateError
from duet.schema import ACTORS, EVIDENCE_TABLES, NODE_TABLES, REL_TABLES, STATUSES


def run_gates(cs, graph, actor):
    if actor not in ACTORS:
        raise GateError('actor', f'{actor!r} may not write; allowed: {", ".join(ACTORS)}')

    ids = [n.id for n in cs.nodes]
    if len(set(ids)) != len(ids):
        raise GateError('integrity', 'duplicate node id in changeset')
    in_cs = {n.id: n for n in cs.nodes}

    def table_of(node_id):
        if node_id in in_cs:
            return in_cs[node_id].table
        found = graph.node(node_id)
        return found[0] if found else None

    def props_of(node_id):
        if node_id in in_cs:
            return in_cs[node_id].props
        found = graph.node(node_id)
        return found[1] if found else {}

    for n in cs.nodes:
        if n.table not in NODE_TABLES:
            raise GateError('integrity', f'unknown table {n.table}')
    for e in cs.edges:
        a, b = table_of(e.src), table_of(e.dst)
        if a is None or b is None:
            raise GateError('integrity', f'{e.rel} {e.src} -> {e.dst}: endpoint does not exist')
        if (a, b) not in REL_TABLES.get(e.rel, []):
            raise GateError('integrity', f'{e.rel} does not connect {a} to {b}')

    supported = defaultdict(list)
    corrected = set()
    committed = set()
    for e in cs.edges:
        if e.rel == 'SUPPORTED_BY':
            supported[e.src].append(e.dst)
        elif e.rel == 'CORRECTS':
            corrected.add(e.dst)
        elif e.rel == 'COMMITTED_BY':
            committed.add(e.src)

    for n in cs.nodes:
        if n.table in EVIDENCE_TABLES:
            sources = supported.get(n.id) or [t[2] for t in graph.neighbors(n.id, 'SUPPORTED_BY', 'out')]
            if not sources:
                raise GateError('evidence', f'{n.id} has no SUPPORTED_BY source')
            if n.table == 'Claim':
                status = n.props.get('status')
                if status not in STATUSES:
                    raise GateError('status', f'{n.id}: status {status!r} not in {STATUSES}')
                if status == 'confirmed':
                    kinds = {props_of(s).get('kind') for s in sources}
                    has_feedback = n.id in corrected or bool(graph.neighbors(n.id, 'CORRECTS', 'in'))
                    if 'statement' not in kinds and not has_feedback:
                        raise GateError('status', f'{n.id}: confirmed needs a statement source or a correction')
        if n.table == 'Commitment':
            owner_known = n.props.get('owner_known')
            if not isinstance(owner_known, bool):
                raise GateError('commitment', f'{n.id}: owner_known must be true or false')
            has_owner = n.id in committed or bool(graph.neighbors(n.id, 'COMMITTED_BY', 'out'))
            if owner_known != has_owner:
                raise GateError('commitment', f'{n.id}: owner_known={owner_known} but COMMITTED_BY present={has_owner}')

    def scope_of(node_id):
        return props_of(node_id).get('scope') or 'tracked'

    for e in cs.edges:
        if scope_of(e.src) == 'tracked' and scope_of(e.dst) == 'private':
            raise GateError('scope', f'{e.rel} {e.src} -> {e.dst}: tracked node may not link to private node')
```

```python
    # the one write path
    def apply(self, changeset, actor, now=None):
        when = now or now_iso()
        try:
            run_gates(changeset, self, actor)
        except GateError as err:
            self.journal('rejected', actor, changeset, when, gate=err.gate, message=str(err))
            raise
        tables = {n.id: n.table for n in changeset.nodes}
        self.execute('BEGIN TRANSACTION')
        try:
            for node_id in changeset.retire:
                self.execute('MATCH (n {id: $id}) DETACH DELETE n', {'id': node_id})
            for node in changeset.nodes:
                self._upsert_node(node, when)
            for edge in changeset.edges:
                src_table = tables.get(edge.src) or self.node(edge.src)[0]
                dst_table = tables.get(edge.dst) or self.node(edge.dst)[0]
                self._upsert_edge(edge, src_table, dst_table, when)
            self.execute('COMMIT')
        except Exception:
            self.execute('ROLLBACK')
            raise
        self.journal('applied', actor, changeset, when)

    def _upsert_node(self, node, when):
        props = dict(node.props)
        props.setdefault('scope', 'tracked')
        props.setdefault('owner', 'karteek')
        props.pop('id', None)
        for reserved in ('created', 'updated', 'history'):
            props.pop(reserved, None)
        existing = self.node(node.id)
        if existing is None:
            keys = ', '.join(f'{k}: ${k}' for k in props)
            self.execute(f'CREATE (n:{node.table} {{id: $id, created: $now, updated: $now, history: [], {keys}}})',
                         dict(props, id=node.id, now=when))
            return
        old = existing[1]
        changed = {k: old.get(k) for k in props if old.get(k) != props[k]}
        sets = ', '.join(f'n.{k} = ${k}' for k in props)
        self.execute(f'MATCH (n:{node.table} {{id: $id}}) SET {sets}, n.updated = $now', dict(props, id=node.id, now=when))
        if changed:
            self.execute(f'MATCH (n:{node.table} {{id: $id}}) SET n.history = list_append(n.history, $h)',
                         {'id': node.id, 'h': json.dumps(changed, sort_keys=True, default=str)})

    def _upsert_edge(self, edge, src_table, dst_table, when):
        self.execute(f'MATCH (a:{src_table} {{id: $a}}), (b:{dst_table} {{id: $b}}) '
                     f'MERGE (a)-[r:{edge.rel}]->(b) SET r.locator = $locator, r.created = coalesce(r.created, $now)',
                     {'a': edge.src, 'b': edge.dst, 'locator': edge.locator or '', 'now': when})

    def journal_entries(self):
        if not self.paths.journal.exists():
            return []
        return [json.loads(line) for line in self.paths.journal.read_text().splitlines() if line.strip()]
```

```python
    line: int


@dataclass
class FeedbackCallout:
    date: str
    claim_target: tuple | None
    scope: str
    applied: bool
    text: str
    line: int


@dataclass
class Page:
    path: str
    hash: str
    frontmatter: dict
    title: str
    links: list = field(default_factory=list)
    claims: list = field(default_factory=list)
    feedback: list = field(default_factory=list)


def slugify(text):
    text = re.sub(r"['’]", '', text.lower())
    text = re.sub(r'[^a-z0-9\s-]', '', text)
    return re.sub(r'\s', '-', text.strip())


def resolve_link(page_path, target):
    if re.match(r'^[a-z][a-z0-9+.-]*:', target) or target.startswith('#'):
        return None, None
    path, _, fragment = target.partition('#')
    if not path.endswith('.md'):
        return None, None
    resolved = posixpath.normpath(posixpath.join(posixpath.dirname(page_path), path))
    if resolved.startswith('..'):
        return None, None
    return resolved, fragment or None


def _parse_frontmatter(lines):
    if not lines or lines[0].strip() != '---':
        return {}, 0
    data = {}
    for i, line in enumerate(lines[1:], start=1):
        if line.strip() == '---':
            return data, i + 1
        if ':' not in line:
            continue
        key, _, value = line.partition(':')
        value = value.strip()
        if value.startswith('[') and value.endswith(']'):
            data[key.strip()] = [v.strip().strip('"').strip("'") for v in value[1:-1].split(',') if v.strip()]
        else:
            data[key.strip()] = value.strip('"').strip("'")
    return {}, 0


def _links_in(text, page_path, line_no):
    out = []
    for m in LINK_RE.finditer(text):
        target, fragment = resolve_link(page_path, m.group(2))
        out.append(Link(m.group(1), target, fragment, line_no))
    return out


def parse_page(vault, rel_path):
    raw = (vault / rel_path).read_bytes()
    lines = raw.decode('utf-8', errors='replace').splitlines()
    frontmatter, start = _parse_frontmatter(lines)
    page = Page(path=rel_path, hash=hashlib.sha256(raw).hexdigest(), frontmatter=frontmatter, title=rel_path)

    sections = []  # (heading, line_no, [lines])
    current = None
    i = start
    while i < len(lines):
        line = lines[i]
        heading = HEADING_RE.match(line)
        if heading:
            if len(heading.group(1)) == 1 and page.title == rel_path:
                page.title = heading.group(2)
            current = (heading.group(2), i + 1, [])
            sections.append(current)
        elif CALLOUT_RE.match(line):
            page.feedback.append(_parse_callout(lines, i, rel_path))
        elif current is not None:
            current[2].append(line)
        page.links.extend(_links_in(line, rel_path, i + 1))
        i += 1

    for heading, line_no, body in sections:
        fields, evidence, supersedes = {}, [], []
        for offset, line in enumerate(body):
            m = FIELD_RE.match(line)
            if not m:
                continue
            key = m.group(1).strip().lower()
            fields[key] = m.group(2).strip()
            if key == 'evidence':
                evidence = [l for l in _links_in(m.group(2), rel_path, line_no + offset + 1) if l.target]
            if key == 'supersedes':
                supersedes = [l for l in _links_in(m.group(2), rel_path, line_no + offset + 1) if l.target]
        if 'claim' in fields:
            page.claims.append(ClaimBlock(heading, slugify(heading), fields, evidence, supersedes, line_no))
    return page


def _parse_callout(lines, start, rel_path):
    date = CALLOUT_RE.match(lines[start]).group(1).strip()
    claim_target, scope, applied, text = None, '', False, []
    for line in lines[start + 1:]:
        if not line.startswith('>'):
            break
        content = line[1:].strip()
        key, sep, value = content.partition(':')
        key = key.strip().lower()
        if sep and key == 'claim':
            links = [l for l in _links_in(value, rel_path, 0) if l.target]
            claim_target = (links[0].target, links[0].fragment) if links else None
        elif sep and key == 'scope':
            scope = value.strip()
        elif sep and key == 'applied':
            applied = value.strip().lower() in ('yes', 'true')
        else:
            text.append(content)
    return FeedbackCallout(date, claim_target, scope, applied, ' '.join(text).strip(), start + 1)
```

```python
"""Entity-Context-Linking: vault pages in, gated changesets out, one report per run."""
import json
from pathlib import Path

from duet.changeset import Changeset, Edge, GateError, Node
from duet.graph import now_iso
from duet.parse import parse_page, slugify

EXCLUDED_PREFIXES = ('.obsidian/', '.git/', 'private/views/', 'private/graph/', 'private/duet/')
KARTEEK = 'person:karteek'


def iter_pages(vault):
    vault = Path(vault)
    out = []
    for path in sorted(vault.rglob('*.md')):
        rel = path.relative_to(vault).as_posix()
        if rel.startswith(EXCLUDED_PREFIXES) or '/.' in f'/{rel}':
            continue
        out.append(rel)
    return out


def scope_of(rel_path):
    return 'private' if rel_path.startswith('private/') else 'tracked'


def doc_id(rel_path):
    return f'doc:{rel_path}'


def stem(rel_path):
    return rel_path[:-3] if rel_path.endswith('.md') else rel_path


def claim_id(rel_path, slug):
    return f'claim:{stem(rel_path)}:{slug}'


def source_id(rel_path):
    return f'source:{Path(rel_path).stem}'


def _common(page, label, locator=None):
    return {'label': label, 'scope': scope_of(page.path), 'owner': 'karteek',
            'source_locator': locator or page.path, 'defined_in': page.path}


def documents_changeset(pages):
    cs = Changeset()
    for page in pages:
        cs.nodes.append(Node('Document', doc_id(page.path), dict(
            _common(page, page.title), path=page.path, hash=page.hash,
            frontmatter_type=page.frontmatter.get('type', ''))))
    return cs


def page_changeset(page, known_claims, known_docs):
    """Everything a page defines besides its Document node, plus its edges."""
    cs = Changeset()
    kind = page.frontmatter.get('type', '')
    if kind == 'personal-source' and page.path.startswith('sources/'):
        cs.nodes.append(Node('Source', source_id(page.path), dict(
            _common(page, page.title), kind='statement', date=page.frontmatter.get('date', ''),
            origin=page.frontmatter.get('origin', ''), provider='')))
    if kind == 'knowledge':
        author = page.frontmatter.get('author', '')
        cid = f'concept:{Path(page.path).stem}'
        cs.nodes.append(Node('Concept', cid, dict(_common(page, page.title), author=author, owner=author or 'unknown')))
        cs.edges.append(Edge('DERIVED_FROM', cid, doc_id(page.path), page.path))
        if author:
            pid = f'person:{slugify(author)}'
            cs.nodes.append(Node('Person', pid, dict(_common(page, author), is_karteek=False, role='author', owner=author)))
            cs.edges.append(Edge('ATTRIBUTED_TO', cid, pid, page.path))
    if page.claims:
        cs.nodes.append(Node('Person', KARTEEK, {'label': 'Karteek', 'scope': 'tracked', 'owner': 'karteek',
                                                 'is_karteek': True, 'role': 'Field CTO', 'defined_in': ''}))
    for block in page.claims:
        cid = claim_id(page.path, block.slug)
        locator = f'{page.path}#{block.slug}'
        cs.nodes.append(Node('Claim', cid, dict(
            _common(page, block.fields.get('claim', block.heading), locator),
            status=block.fields.get('status', 'unknown').split()[0].lower(),
            scope_text=block.fields.get('scope', ''), recorded=block.fields.get('recorded', ''),
            review_by=block.fields.get('review by', ''), exceptions=block.fields.get('exceptions / conflicts', ''))))
        cs.edges.append(Edge('ABOUT', cid, KARTEEK, locator))
        for link in block.evidence_links:
            if link.target.startswith('sources/'):
                cs.edges.append(Edge('SUPPORTED_BY', cid, source_id(link.target), link.fragment or link.target))
        for link in block.supersedes_links:
            target = claim_id(link.target, link.fragment or '')
            if target in known_claims:
                cs.edges.append(Edge('SUPERSEDES', cid, target, locator))
    for n, fb in enumerate(page.feedback, start=1):
        fid = f'feedback:{stem(page.path)}:{n}'
        cs.nodes.append(Node('Feedback', fid, dict(
            _common(page, fb.text[:80] or f'Feedback {n}', f'{page.path}:{fb.line}'),
            date=fb.date, scope_text=fb.scope, applied=fb.applied)))
        if fb.claim_target and fb.claim_target[1]:
            target = claim_id(fb.claim_target[0], fb.claim_target[1])
            if target in known_claims:
                cs.edges.append(Edge('CORRECTS', fid, target, f'{page.path}:{fb.line}'))
    seen = set()
    for link in page.links:
        if not link.target or link.target == page.path:
            continue
        if link.target in known_docs and ('LINKS_TO', link.target) not in seen:
            cs.edges.append(Edge('LINKS_TO', doc_id(page.path), doc_id(link.target), f'{page.path}:{link.line}'))
            seen.add(('LINKS_TO', link.target))
        if link.fragment:
            target = claim_id(link.target, link.fragment)
            if target in known_claims and ('MENTIONS', target) not in seen:
                cs.edges.append(Edge('MENTIONS', doc_id(page.path), target, f'{page.path}:{link.line}'))
                seen.add(('MENTIONS', target))
    return cs


def index(vault, graph, force=False):
    vault = Path(vault)
    p = graph.paths
    state = json.loads(p.index_state.read_text()) if p.index_state.exists() else {}
    report = {'started': now_iso(), 'parsed': [], 'unchanged': [], 'skipped': [], 'rejected': [], 'retired': [], 'broken_links': []}

    pages, new_state = [], {}
    for rel in iter_pages(vault):
        try:
            pages.append(parse_page(vault, rel))
        except Exception as err:  # unreadable page: keep its nodes, report it
            report['skipped'].append({'path': rel, 'reason': str(err)})
            if rel in state:
                new_state[rel] = state[rel]
    by_path = {page.path: page for page in pages}
    known_docs = set(by_path)
    known_claims = {claim_id(page.path, c.slug) for page in pages for c in page.claims}

    for rel in state:
        if rel not in by_path and rel not in {s['path'] for s in report['skipped']}:
            retire = [row[0] for row in graph.rows('MATCH (n) WHERE n.defined_in = $p RETURN n.id', {'p': rel})]
            graph.apply(Changeset(retire=retire), 'indexer')
            report['retired'] += retire

    changed = [page for page in pages if force or state.get(page.path) != page.hash]
    for page in pages:
        if page not in changed:
            report['unchanged'].append(page.path)
            new_state[page.path] = page.hash
    if changed:
        graph.apply(documents_changeset(changed), 'indexer')

    for page in changed:
        for link in page.links:
            if link.target and link.target not in known_docs:
                report['broken_links'].append({'path': page.path, 'line': link.line, 'target': link.target})
        cs = page_changeset(page, known_claims, known_docs)
        defined_now = {n.id for n in cs.nodes} | {doc_id(page.path)}
        previous = [row[0] for row in graph.rows('MATCH (n) WHERE n.defined_in = $p RETURN n.id', {'p': page.path})]
        cs.retire = [node_id for node_id in previous if node_id not in defined_now]
        try:
            graph.apply(cs, 'indexer')
        except GateError as err:
            report['rejected'].append({'path': page.path, 'gate': err.gate, 'message': str(err)})
            continue
        report['parsed'].append(page.path)
        report['retired'] += cs.retire
        new_state[page.path] = page.hash

    report['finished'] = now_iso()
    p.graph_dir.mkdir(parents=True, exist_ok=True)
    p.index_state.write_text(json.dumps(new_state, indent=2, sort_keys=True) + '\n')
    p.index_report.write_text(json.dumps(report, indent=2) + '\n')
    return report
```

```python
import posixpath
from datetime import date

from duet.graph import now_iso

VIEWS_DIR = 'private/views'


def rel(target):
    """Vault-relative target as a link from private/views/."""
    return posixpath.relpath(target, VIEWS_DIR)


def header(title, p):
    return [f'# {title}', '', f'Generated {now_iso()} by `python -m duet project` · scope: tracked and private · '
            'every line links its claim and its source. Edit the linked pages, not this file.', '']


def claim_rows(graph):
    rows = graph.rows(
        'MATCH (c:Claim) OPTIONAL MATCH (c)-[:SUPPORTED_BY]->(s:Source) '
        'RETURN c.id, c.label, c.status, c.scope_text, c.recorded, c.review_by, c.source_locator, c.scope, collect(s.defined_in) '
        'ORDER BY c.recorded, c.id')
    return [dict(id=r[0], label=r[1], status=r[2], scope_text=r[3], recorded=r[4], review_by=r[5], locator=r[6], scope=r[7],
                 sources=[s for s in r[8] if s]) for r in rows]


def claim_line(c):
    sources = ' '.join(f'[{s}]({rel(s)})' for s in c['sources']) or 'no source'
    page, _, anchor = c['locator'].partition('#')
    claim_link = f'[{c["locator"]}]({rel(page)}#{anchor})' if anchor else f'[{c["locator"]}]({rel(page)})'
    return (f'- **{c["label"]}** · status: {c["status"]} · scope: {c["scope_text"] or "unspecified"} · recorded {c["recorded"] or "unknown"}'
            f' · review by: {c["review_by"] or "not set"} · source: {sources} · claim: {claim_link}')


def feedback_for(graph, claim_id):
    return graph.rows('MATCH (f:Feedback)-[:CORRECTS]->(c:Claim {id: $id}) RETURN f.date, f.label, f.source_locator ORDER BY f.date', {'id': claim_id})


def render_me(graph, p):
    claims = claim_rows(graph)
    lines = header('What the graph knows about me', p)
    for status, title in (('confirmed', 'Confirmed'), ('hypothesis', 'Hypotheses'), ('unknown', 'Unknown'), ('superseded', 'Superseded')):
        group = [c for c in claims if c['status'] == status]
        lines += [f'## {title} ({len(group)})', '']
        lines += [claim_line(c) for c in group] or ['None recorded yet.']
        lines.append('')
    return '\n'.join(lines)


def render_review_queue(graph, p):
    lines = header('Review queue: hypotheses awaiting your confirmation', p)
    group = [c for c in claim_rows(graph) if c['status'] == 'hypothesis']
    if not group:
        lines.append('None recorded yet.')
    for c in group:
        lines.append(claim_line(c))
        for fdate, ftext, floc in feedback_for(graph, c['id']):
            lines.append(f'  - feedback {fdate}: {ftext} ([{floc}]({rel(floc.split(":")[0])}))')
    lines += ['', 'To confirm a hypothesis, add a `> [!feedback]` callout in `memory/feedback.md` linking the claim, then run `python -m duet index`.']
    return '\n'.join(lines) + '\n'


def render_stale(graph, p):
    today = date.today().isoformat()
    lines = header('Stale: claims past their review date', p)
    stale = []
    for c in claim_rows(graph):
        value = (c['review_by'] or '')[:10]
        if len(value) == 10 and value[4] == '-' and value < today:
            stale.append(c)
    lines += [claim_line(c) for c in stale] or ['None recorded yet.']
    return '\n'.join(lines) + '\n'


def render_simple_list(graph, p, title, table):
    lines = header(title, p)
    rows = graph.rows(f'MATCH (n:{table}) OPTIONAL MATCH (n)-[:SUPPORTED_BY]->(s:Source) RETURN n.label, n.status, n.source_locator, collect(s.defined_in) ORDER BY n.id')
    for label, status, locator, sources in rows:
        src = ' '.join(f'[{s}]({rel(s)})' for s in sources if s) or 'no source'
        lines.append(f'- **{label}** · status: {status or "n/a"} · source: {src} · locator: {locator}')
    if not rows:
        lines.append('None recorded yet.')
    return '\n'.join(lines) + '\n'


def render_index_report(graph, p):
    lines = header('Last index run', p)
    if not p.index_report.exists():
        return '\n'.join(lines + ['No index run recorded.']) + '\n'
    report = json.loads(p.index_report.read_text())
    lines += [f'- started: {report.get("started")}', f'- finished: {report.get("finished")}',
              f'- parsed: {len(report.get("parsed", []))}', f'- unchanged: {len(report.get("unchanged", []))}',
              f'- rejected: {len(report.get("rejected", []))}', f'- skipped: {len(report.get("skipped", []))}',
              f'- retired: {len(report.get("retired", []))}', f'- broken links: {len(report.get("broken_links", []))}', '']
    for item in report.get('rejected', []):
        lines.append(f'- rejected [{item["path"]}]({rel(item["path"])}): {item["message"]}')
    for item in report.get('skipped', []):
        lines.append(f'- skipped {item["path"]}: {item["reason"]}')
    for item in report.get('broken_links', []):
        lines.append(f'- broken link in [{item["path"]}]({rel(item["path"])}) line {item["line"]}: `{item["target"]}`')
    return '\n'.join(lines) + '\n'


def project(graph, p):
    p.views.mkdir(parents=True, exist_ok=True)
    outputs = {
        'me.md': render_me(graph, p),
        'review-queue.md': render_review_queue(graph, p),
        'stale.md': render_stale(graph, p),
        'commitments.md': render_simple_list(graph, p, 'Commitments', 'Commitment'),
        'decisions.md': render_simple_list(graph, p, 'Decisions', 'Decision'),
        'index-report.md': render_index_report(graph, p),
    }
    for name, text in outputs.items():
        (p.views / name).write_text(text)
    return sorted(outputs)
```

```python
"""Read-only MCP tools over the graph, for Claude Code, Claudian and the runner."""
import json
import re

from mcp.server.mcpserver import MCPServer

from duet.graph import Graph
from duet.schema import NODE_TABLES, REL_TABLES

WRITE_WORDS = re.compile(r'\b(CREATE|MERGE|SET|DELETE|DETACH|DROP|COPY|INSTALL|LOAD|ALTER|BEGIN|COMMIT|ROLLBACK|IMPORT|EXPORT|ATTACH|USE)\b', re.I)
READ_START = re.compile(r'^\s*(MATCH|OPTIONAL\s+MATCH|RETURN|WITH|UNWIND|CALL\s+SHOW_)', re.I)


def guard_read_only(cypher):
    if WRITE_WORDS.search(cypher) or not READ_START.match(cypher):
        raise ValueError('read-only: only MATCH/RETURN/WITH/UNWIND/CALL SHOW_* queries are allowed')
    return cypher


def _rows_as_dicts(result):
    columns = result.get_column_names()
    out = []
    while result.has_next():
        out.append(dict(zip(columns, result.get_next())))
    return out


def build_server(p):
    graph = Graph.open(p, read_only=True)
    server = MCPServer('duet', instructions=(
        'Read-only view of the Karteek Duet knowledge graph. Every Claim carries a status (confirmed, hypothesis, unknown, superseded) '
        'and SUPPORTED_BY edges to its Source; cite the source locator when you use a claim. Nothing here writes to the graph.'))
    server.close_graph = graph.close

    @server.tool()
    def get_entity(id: str) -> dict:
        """One node by id with its properties and every connection grouped by relation."""
        found = graph.node(id)
        if not found:
            raise ValueError(f'no node {id}')
        table, props = found
        connections = {}
        for rel, other_table, other_id, locator in graph.neighbors(id, None, 'out'):
            connections.setdefault(rel, []).append({'id': other_id, 'type': other_table, 'locator': locator, 'direction': 'out'})
        for rel, other_table, other_id, locator in graph.neighbors(id, None, 'in'):
            connections.setdefault(rel, []).append({'id': other_id, 'type': other_table, 'locator': locator, 'direction': 'in'})
        return {'id': id, 'type': table, 'props': props, 'connections': connections}

    @server.tool()
    def related(id: str, relation: str, direction: str = 'both', type: str | None = None) -> list[dict]:
        """Nodes connected to id by one relation (see schema relations), optionally filtered by node type."""
        if relation not in REL_TABLES:
            raise ValueError(f'unknown relation {relation}; one of {sorted(REL_TABLES)}')
        out = []
        for rel, other_table, other_id, locator in graph.neighbors(id, relation, direction):
            if type and other_table != type:
                continue
            out.append({'id': other_id, 'type': other_table, 'locator': locator})
        return out

    @server.tool()
    def search(text: str, limit: int = 10) -> list[dict]:
        """Case-insensitive substring search over node labels across every table."""
        rows = graph.rows('MATCH (n) WHERE lower(n.label) CONTAINS lower($q) RETURN label(n), n.id, n.label, n.scope ORDER BY n.id LIMIT $limit',
                          {'q': text, 'limit': int(limit)})
        return [{'type': r[0], 'id': r[1], 'label': r[2], 'scope': r[3]} for r in rows]

    @server.tool()
    def claims(status: str | None = None) -> list[dict]:
        """Claims about Karteek, optionally filtered by status, each with its source locators."""
        rows = graph.rows('MATCH (c:Claim) OPTIONAL MATCH (c)-[:SUPPORTED_BY]->(s:Source) '
                          'RETURN c.id, c.label, c.status, c.scope_text, c.recorded, c.review_by, c.source_locator, collect(s.defined_in) ORDER BY c.id')
        out = [{'id': r[0], 'label': r[1], 'status': r[2], 'scope_text': r[3], 'recorded': r[4], 'review_by': r[5],
                'locator': r[6], 'sources': [s for s in r[7] if s]} for r in rows]
        return [c for c in out if status is None or c['status'] == status]

    @server.tool()
    def query(cypher: str) -> list[dict]:
        """Run a read-only Cypher query (MATCH/RETURN/WITH/UNWIND). Writing keywords are refused."""
        return _rows_as_dicts(graph.execute(guard_read_only(cypher)))

    @server.tool()
    def duet_status() -> dict:
        """Node counts per table and the last index run summary."""
        last = {}
        if p.index_report.exists():
            report = json.loads(p.index_report.read_text())
            last = {key: len(report.get(key, [])) for key in ('parsed', 'unchanged', 'rejected', 'skipped', 'retired', 'broken_links')}
            last['finished'] = report.get('finished')
        return {'vault': str(p.vault), 'counts': graph.counts(), 'tables': sorted(NODE_TABLES), 'last_index': last}

    return server


def serve(p):
    server = build_server(p)
    try:
        server.run('stdio')
    finally:
        server.close_graph()
```

```python
def cmd_rebuild(p, args):
    replay = []
    if p.journal.exists():
        for line in p.journal.read_text().splitlines():
            if not line.strip():
                continue
            entry = json.loads(line)
            if entry.get('actor') != 'indexer' and entry.get('result') == 'applied' and 'nodes' in entry.get('changeset', {}):
                replay.append(entry)
    for path in (p.db, p.index_state):
        if path.exists():
            path.unlink()
    for extra in p.graph_dir.glob('duet.lbug.*'):
        extra.unlink()
    replayed = failed = 0
    with Graph.open(p) as graph:
        graph.ensure_schema()
        report = run_index(p.vault, graph, force=True)
        for entry in replay:
            try:
                graph.apply(Changeset.from_json(entry['changeset']), entry['actor'])
                replayed += 1
            except GateError:
                failed += 1
        run_project(graph, p)
    print(f'rebuilt: parsed={len(report["parsed"])} rejected={len(report["rejected"])} replayed={replayed} replay_failed={failed}')
    return 1 if report['rejected'] or failed else 0
```
